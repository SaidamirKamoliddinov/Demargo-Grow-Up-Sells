Ilova/dasturlash haqida ko'proq bilib oling 
import { DemargoClient } from '@demargo/sdk';

const client = new DemargoClient({ apiKey: process.env.DEMARGO_KEY });

// Create instant freight dispatch
const shipment = await client.shipments.create({
  origin: 'USLAX', // Port of Los Angeles
  destination: 'NLTAM', // Port of Rotterdam
  cargoType: 'OCEAN_FCL_40HC',
  autoCustoms: true
});

console.log(`Shipment active: ${shipment.trackingId}`);
from demargo import DemargoSDK

client = DemargoSDK(api_key="dm_live_secret_9981")

# Query instant freight routing & tariffs
routing = client.freight.get_optimal_route(
    origin="USLAX",
    destination="NLTAM",
    weight_kg=12500
)

print(f"Estimated ETA: {routing.eta_days} days")
